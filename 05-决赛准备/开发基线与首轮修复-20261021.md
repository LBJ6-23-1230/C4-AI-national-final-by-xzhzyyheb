```powershell
$devecoDir = 'D:\develop\DevEco Studio'
$env:DEVECO_SDK_HOME = "$devecoDir\sdk"
$env:JAVA_HOME = "$devecoDir\jbr"
$env:Path = "$devecoDir\tools\node;$devecoDir\tools\ohpm\bin;" + $env:Path
```
